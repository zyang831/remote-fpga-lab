cmake_minimum_required(VERSION 3.20)
project(dla-server C CXX)
find_package(OpenVINO CONFIG REQUIRED)
find_package(OpenCV COMPONENTS core imgcodecs imgproc REQUIRED)
find_package(Threads REQUIRED)
add_executable(dla-server main.cpp cam/mmio.c cam/altera_i2c.c cam/imx519.c cam/cam_msgdma.c)
target_compile_features(dla-server PRIVATE cxx_std_17)
target_include_directories(dla-server PRIVATE ${DLA_PLUGIN_INC})
target_link_libraries(dla-server PRIVATE openvino::runtime ${OpenCV_LIBRARIES} ${JPEG_LIB} Threads::Threads)
set_target_properties(dla-server PROPERTIES BUILD_RPATH "\$ORIGIN")
